=== Chat with a tool call
--> POST /ollama/api/chat
{
  "model" : "llama3.2",
  "messages" : [ {
    "role" : "user",
    "content" : "What is the weather in Athens?"
  } ],
  "stream" : false,
  "tools" : [ {
    "type" : "function",
    "function" : {
      "name" : "get_weather",
      "description" : "Returns the weather in a city",
      "parameters" : {
        "type" : "object",
        "properties" : {
          "city" : {
            "type" : "string"
          }
        },
        "required" : [ "city" ]
      }
    }
  } ]
}
<-- 200 [application/json]
{
  "model" : "llama3.2",
  "created_at" : "2026-01-01T00:00:00Z",
  "message" : {
    "role" : "assistant",
    "content" : "",
    "tool_calls" : [ {
      "function" : {
        "name" : "get_weather",
        "arguments" : {
          "city" : "Athens"
        }
      }
    } ]
  },
  "done" : true,
  "done_reason" : "stop",
  "total_duration" : 0,
  "load_duration" : 0,
  "prompt_eval_count" : 0,
